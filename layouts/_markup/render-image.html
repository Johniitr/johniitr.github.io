{{- /* Markdown images: page-bundle images are resized and served as WebP with a phone-sized variant. */ -}} {{- $src := .Destination -}} {{- $res := .Page.Resources.Get $src -}} {{- if and $res (in (slice "image/jpeg" "image/png" "image/webp") $res.MediaType.Type) -}} {{- $w := cond (gt $res.Width 1440) 1440 $res.Width -}} {{- $big := $res.Resize (printf "%dx webp q82" $w) -}} {{- $small := $big -}} {{- if gt $w 720 }}{{ $small = $res.Resize "720x webp q80" }}{{ end -}} {{- else -}} {{- end -}}
